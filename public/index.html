const { app, BrowserWindow, session, desktopCapturer, ipcMain, powerSaveBlocker } = require('electron'); const path = require('path'); const { spawn } = require('child_process'); const { AudioCaptureHelper } = require('./audio-helper'); const { listAudioProcesses } = require('./list-audio-processes'); // Troque aqui se o endereço do seu DSpeak mudar. const SITE_URL = 'https://dspeak.onrender.com'; let mainWindow = null; let pickerWindow = null; let powerSaveBlockerId = null; // ---------- Captura de áudio por PROCESSO específico ---------- // Roda num programa SEPARADO (dspeak-audio.exe), de propósito — se ele travar/crashar, // só ele morre, o DSpeak inteiro continua funcionando normal (a gente só reinicia esse // programinha sozinho, com um número limitado de tentativas). const audioHelper = new AudioCaptureHelper(); function normalizeTitle(t) { return String(t || '') .replace(/[^a-zA-Z0-9\s]/g, '') .replace(/\s+/g, ' ') .trim() .toLowerCase(); } // ---------- Áudio automático pro modo "Janelas" ---------- // Quando a pessoa escolhe uma janela específica pro vídeo, a gente já sabe pra qual // programa aquela janela pertence — não faz sentido perguntar de novo qual áudio usar. // O identificador da fonte de vídeo (no Windows) vem no formato "window::0" — // extraímos o HWND dali e usamos o PowerShell (mesma técnica do list-audio-processes.js) // pra descobrir qual processo é dono dessa janela. function parseHwndFromWindowSourceId(sourceId) { const match = /^window:(\d+):/.exec(String(sourceId || '')); return match ? match[1] : null; } function getPidForHwnd(hwnd) { return new Promise((resolve) => { const script = `Get-Process | Where-Object { $_.MainWindowHandle -eq ${hwnd} } | Select-Object -First 1 -ExpandProperty Id`; const ps = spawn('powershell.exe', ['-NoProfile', '-NonInteractive', '-Command', script]); let out = ''; ps.stdout.on('data', (d) => { out += d.toString(); }); ps.on('error', () => resolve(null)); ps.on('close', () => { const pid = parseInt(out.trim(), 10); resolve(Number.isFinite(pid) && pid > 0 ? pid : null); }); }); } // Recebe a fonte de vídeo escolhida (uma janela) e devolve o audioMode pronto — // { mode: 'process', processId } se achou o dono, ou { mode: 'none' } se não achou // (não trava o compartilhamento por causa disso, só vai sem áudio automático). async function resolveAutoWindowAudioMode(source) { const hwnd = parseHwndFromWindowSourceId(source && source.id); if (!hwnd) { console.warn('[DSpeak] Não consegui identificar o HWND da janela escolhida — compartilhando sem áudio automático.'); return { mode: 'none' }; } const pid = await getPidForHwnd(hwnd); if (!pid) { console.warn(`[DSpeak] Não encontrei o processo dono da janela (HWND ${hwnd}) — compartilhando sem áudio automático.`); return { mode: 'none' }; } console.log(`[DSpeak] Janela escolhida pertence ao processo ${pid} — usando áudio dele automaticamente.`); return { mode: 'process', processId: pid }; } // Busca a lista de janelas/telas (com miniatura) pro vídeo da transmissão. async function getVideoSources() { const sources = await desktopCapturer.getSources({ types: ['window', 'screen'], thumbnailSize: { width: 400, height: 300 }, fetchWindowIcons: true }); return sources.map(s => ({ raw: s, id: s.id, name: s.name, type: s.id.startsWith('screen') ? 'screen' : 'window', thumbnailDataUrl: s.thumbnail.isEmpty() ? null : s.thumbnail.toDataURL() })); } function createMainWindow() { mainWindow = new BrowserWindow({ width: 1400, height: 900, minWidth: 900, minHeight: 600, icon: path.join(__dirname, 'build', 'icon.ico'), autoHideMenuBar: true, backgroundColor: '#0c0d10', webPreferences: { preload: path.join(__dirname, 'preload.js'), contextIsolation: true, nodeIntegration: false, sandbox: false, // Evita o Chromium desacelerar os temporizadores da página quando minimizada — // sem isso, o "sinal de vida" da conexão de voz/chat também desacelerava, e o // servidor achava que você tinha caído. backgroundThrottling: false } }); mainWindow.loadURL(SITE_URL); mainWindow.webContents.setBackgroundThrottling(false); // Impede o Windows de suspender o app quando minimizado (mesma técnica usada por // apps de chamada de vídeo/voz, tipo Discord/Teams). Guardamos o ID retornado pra // nunca empilhar blockers achando que esquecemos de parar o anterior. if (powerSaveBlockerId === null || !powerSaveBlocker.isStarted(powerSaveBlockerId)) { powerSaveBlockerId = powerSaveBlocker.start('prevent-app-suspension'); } // Liga o F12 pra abrir/fechar o Console de Desenvolvedor — não vem religado por // padrão no Electron. mainWindow.webContents.on('before-input-event', (event, input) => { if (input.key === 'F12' && input.type === 'keyDown') { mainWindow.webContents.toggleDevTools(); } }); mainWindow.on('closed', () => { mainWindow = null; }); } // ---------- IPC: iniciar/parar captura de áudio de um processo específico ---------- ipcMain.handle('start-process-audio-capture', (event, processId) => { return new Promise((resolve) => { let resolved = false; let chunkCount = 0; let detectedSampleRate = null; let detectedChannels = null; let detectedSampleFormat = null; audioHelper.start(processId, { onSampleRate: (hz) => { detectedSampleRate = hz; console.log(`[DSpeak] Taxa de amostragem nativa detectada pro PID ${processId}: ${hz} Hz.`); }, onChannels: (n) => { detectedChannels = n; console.log(`[DSpeak] Número de canais nativo detectado pro PID ${processId}: ${n}.`); }, onSampleFormat: (fmt) => { detectedSampleFormat = fmt; console.log(`[DSpeak] Formato de amostra nativo detectado pro PID ${processId}: ${fmt}.`); }, onData: (chunk) => { chunkCount++; if (chunkCount === 1) { console.log(`[DSpeak] Primeiro pedaço de áudio chegou do PID ${processId} (${chunk.length} bytes) — captura funcionando.`); if (!resolved) { resolved = true; resolve({ ok: true, sampleRate: detectedSampleRate, channels: detectedChannels, sampleFormat: detectedSampleFormat }); } } if (chunkCount % 100 === 0) { // Confere se o pedaço é só silêncio (todos os bytes zero) ou tem som de // verdade — isso diz se a captura está realmente pegando áudio ou só // mandando silêncio de reserva o tempo todo. const isSilent = chunk.every(b => b === 0); console.log(`[DSpeak] Pedaço #${chunkCount} (${chunk.length} bytes) — ${isSilent ? 'SILÊNCIO' : 'TEM SOM'}`); } if (mainWindow && !mainWindow.isDestroyed()) { mainWindow.webContents.send('process-audio-chunk', chunk); } }, onError: (message, permanent) => { console.error(`[DSpeak] Captura de áudio por processo falhou (permanente=${permanent}):`, message); if (mainWindow && !mainWindow.isDestroyed()) { mainWindow.webContents.send('process-audio-error', { message, permanent }); } if (!resolved) { resolved = true; resolve({ ok: false, error: message }); } } }); // Se depois de um tempo não chegou nem o primeiro pedaço nem um erro, libera a // Promise mesmo assim (evita travar quem está esperando indefinidamente). setTimeout(() => { if (!resolved) { resolved = true; resolve({ ok: true, sampleRate: detectedSampleRate, channels: detectedChannels, sampleFormat: detectedSampleFormat, note: 'sem confirmação de dados ainda' }); } }, 2000); }); }); ipcMain.handle('stop-process-audio-capture', () => { audioHelper.stop(); return { ok: true }; }); ipcMain.handle('list-audio-processes', async () => { return await listAudioProcesses(); }); // ---------- Captura de tela nativa ---------- // O Windows não tem (ainda) um seletor de tela "automático" do sistema operacional // como o macOS tem — então construímos nossa própria telinha de escolha. function setupDisplayMediaHandler() { session.defaultSession.setDisplayMediaRequestHandler(async (request, callback) => { try { const sources = await getVideoSources(); const chosen = await showSourcePicker(sources); if (!chosen || !chosen.source) { callback({}); return; } const result = { video: chosen.source.raw }; const audioMode = chosen.audioMode || { mode: 'none' }; if (audioMode.mode === 'system') { result.audio = 'loopback'; } else if (audioMode.mode === 'process' && audioMode.processId) { // Áudio de processo específico não é suportado nativamente pela API do // Electron — avisamos a página pra ela pedir a captura via IPC (usando o // dspeak-audio.exe) e montar a faixa de áudio na hora. if (mainWindow && !mainWindow.isDestroyed()) { mainWindow.webContents.send('use-process-audio', audioMode.processId); } } callback(result); } catch (err) { console.error('Erro ao capturar tela:', err); callback({}); } }, { useSystemPicker: false }); } function showSourcePicker(sources) { return new Promise((resolve) => { if (pickerWindow) { pickerWindow.close(); pickerWindow = null; } pickerWindow = new BrowserWindow({ width: 820, height: 700, minWidth: 700, minHeight: 500, parent: mainWindow, modal: true, resizable: true, minimizable: false, maximizable: false, autoHideMenuBar: true, backgroundColor: '#13151b', icon: path.join(__dirname, 'build', 'icon.ico'), webPreferences: { preload: path.join(__dirname, 'picker-preload.js'), contextIsolation: true, nodeIntegration: false } }); pickerWindow.setMenuBarVisibility(false); pickerWindow.loadFile(path.join(__dirname, 'picker.html')); let settled = false; // Guarda os handlers com nome (não anônimos) pra dar pra remover certinho depois, // não importa por qual caminho a telinha fechou (selecionou, cancelou, ou fechou // no X) — sem isso, cada vez que a telinha abria de novo, os listeners antigos // ficavam acumulando sem nunca serem removidos. const selectHandler = async (event, sourceId, audioMode) => { const found = sources.find(s => s.id === sourceId); if (!found) { finish(null); return; } // Modo "auto" (vem do modo Janelas do picker.html): resolve o PID sozinho a // partir da janela escolhida, em vez de usar o que veio do picker direto. const resolvedAudioMode = (audioMode && audioMode.mode === 'auto') ? await resolveAutoWindowAudioMode(found) : audioMode; finish({ source: found, audioMode: resolvedAudioMode }); }; const cancelHandler = () => finish(null); const cleanupListeners = () => { ipcMain.removeListener('picker-refresh', refreshHandler); ipcMain.removeListener('picker-select', selectHandler); ipcMain.removeListener('picker-cancel', cancelHandler); }; const finish = (result) => { if (settled) return; settled = true; cleanupListeners(); resolve(result); if (pickerWindow) { pickerWindow.close(); } }; const sendSources = (list) => { const serializable = list.map(s => ({ id: s.id, name: s.name, type: s.type, thumbnailDataUrl: s.thumbnailDataUrl })); if (pickerWindow) pickerWindow.webContents.send('sources', serializable); }; // Manda a lista de programas com áudio disponível, à parte da lista de vídeo — // assim dá pra escolher o áudio de um programa mesmo compartilhando a tela // inteira (ou qualquer outra coisa) como vídeo. const sendAudioProcesses = async () => { const list = await listAudioProcesses(); if (pickerWindow) pickerWindow.webContents.send('audio-processes', list); }; // Botão "Atualizar": busca de novo a lista de janelas/telas do zero. const refreshHandler = async () => { try { sources = await getVideoSources(); sendSources(sources); sendAudioProcesses(); } catch (err) { console.error('Erro ao atualizar lista:', err); } }; ipcMain.on('picker-refresh', refreshHandler); pickerWindow.webContents.once('did-finish-load', () => { sendSources(sources); sendAudioProcesses(); }); ipcMain.once('picker-select', selectHandler); ipcMain.once('picker-cancel', cancelHandler); pickerWindow.on('closed', () => { pickerWindow = null; finish(null); }); }); } app.whenReady().then(() => { setupDisplayMediaHandler(); createMainWindow(); app.on('activate', () => { if (BrowserWindow.getAllWindows().length === 0) createMainWindow(); }); }); app.on('window-all-closed', () => { audioHelper.stop(); if (process.platform !== 'darwin') app.quit(); });
